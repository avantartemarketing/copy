{#
  THE TWITTER SET — the tweets in the comms plan: coming soon, announcement, now live, halfway, last chance.

  A tweet is the post's status line with the X handle, the hook on the announcement only, the beneficiary
  named once, and an action line that ends with the link. No features line, no hashtag, no bio.
  Coming Soon names the artist and not the work, as the Instagram tease does. Same deadline form as the posts.
  The same lines live in tool/release-copy.xlsx, sheet Tweets.
#}

{% set named = artist.name ~ ((" (@" ~ xh ~ ")") if xh else "") %}
{% set collab = artist.collab_with if (artist.collab_with is defined and artist.collab_with) else artist.name %}
{% set collab_x = named if collab == artist.name else collab %}
{% set a_unit = edition|a_unit %}
{% set link = release.link if release.link is defined and release.link else "[link]" %}
{% set draw = release.mechanic == "draw" %}
{% set fundraiser = release.fundraiser if release.fundraiser is defined and release.fundraiser else "" %}
{% set series = artworks[0].series if artworks[0].series is defined and artworks[0].series else "" %}
{% set untitled = artworks[0].title == "Untitled" and series %}
{% set work = ("the " ~ series ~ " series") if untitled else artworks|titles_q %}
{% set deadline = "Closes " ~ release.closes_at|ukdate ~ " at " ~ release.closes_at|uktime ~ "." %}
{% set pre_launch = (deadline ~ " Enter the draw: " ~ link) if draw else ("Available for " ~ release.window ~ " from " ~ release.launch_at|uktime ~ " on " ~ release.launch_at|ukdate ~ ". Sign up for updates: " ~ link) %}
{% set open = (deadline ~ " Enter the draw: " ~ link) if draw else (deadline ~ " Buy " ~ a_unit ~ ": " ~ link) %}
